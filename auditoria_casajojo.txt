import com.example.casajojo.navigation.AppNavigation
import com.example.casajojo.ui.theme.CasaJojoTheme



class MainActivity : ComponentActivity() {

    private val requestNotificationPermissionLauncher =
        registerForActivityResult(ActivityResultContracts.RequestPermission()) { _ -> }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
            if (ContextCompat.checkSelfPermission(
                    this,
                    Manifest.permission.POST_NOTIFICATIONS
                ) != PackageManager.PERMISSION_GRANTED
            ) {
                requestNotificationPermissionLauncher.launch(Manifest.permission.POST_NOTIFICATIONS)
            }
        }

        val database = AppDatabase.getInstance(applicationContext)
        val notificationManager = OrderNotificationManager(applicationContext)

        val menuRepository = MenuRepository(database.menuDao())
        val pedidoRepository = PedidoRepository(database.pedidoDao())

        val adminViewModel = ViewModelProvider(
            this,
            AdminViewModelFactory(menuRepository)
        )[AdminViewModel::class.java]

        val operationsViewModel = ViewModelProvider(
            this,
            OperationsViewModelFactory(database.pedidoDao(), notificationManager)
        )[OperationsViewModel::class.java]

        // INSTANCIAR ViewModel DEL CLIENTE
        val clienteViewModel = ViewModelProvider(
            this,
            com.example.casajojo.feature.client.ClienteViewModelFactory(menuRepository, pedidoRepository)
        )[com.example.casajojo.feature.client.ClienteViewModel::class.java]

        setContent {
            CasaJojoTheme {
                AppNavigation(
                    adminViewModel = adminViewModel,
                    operationsViewModel = operationsViewModel,
                    clienteViewModel = clienteViewModel
                )
            }
        }
    }
}

package com.example.casajojo.data.local

import android.content.Context
import androidx.room.Database
import androidx.room.Room
import androidx.room.RoomDatabase
import androidx.room.TypeConverters
import com.example.casajojo.data.model.MenuItem
import com.example.casajojo.data.model.Pedido

@Database(
    entities = [MenuItem::class, Pedido::class],
    version = 1,
    exportSchema = false
)
@TypeConverters(Converters::class)
abstract class AppDatabase : RoomDatabase() {

    abstract fun menuDao(): MenuDao
    abstract fun pedidoDao(): PedidoDao

    companion object {
        @Volatile
        private var INSTANCE: AppDatabase? = null

        fun getInstance(context: Context): AppDatabase {
            return INSTANCE ?: synchronized(this) {
                val instance = Room.databaseBuilder(
                    context.applicationContext,
                    AppDatabase::class.java,
                    "casajojo_database.db"
                ).build()
                INSTANCE = instance
                instance
            }
        }
    }
}

import androidx.room.Delete
import androidx.room.Insert
import androidx.room.Query
import androidx.room.Update
import com.example.casajojo.data.model.MenuItem
import kotlinx.coroutines.flow.Flow

@Dao
interface MenuDao {

    @Query(
        """
        SELECT * FROM menu_items
        WHERE fechaMenu = :fecha
        ORDER BY nombre COLLATE NOCASE ASC
        """
    )
    fun observarPorFecha(fecha: String): Flow<List<MenuItem>>

    @Insert
    suspend fun insertar(item: MenuItem)

    @Update
    suspend fun actualizar(item: MenuItem)

    @Delete
    suspend fun eliminar(item: MenuItem)

    @Query(
        """
        UPDATE menu_items
        SET disponible = :disponible
        WHERE id = :id
        """
    )
    suspend fun actualizarDisponibilidad(
        id: Int,
        disponible: Boolean
    )
}

package com.example.casajojo.data.repository

import com.example.casajojo.data.local.MenuDao
import com.example.casajojo.data.model.MenuItem
import kotlinx.coroutines.flow.Flow

class MenuRepository(
    private val dao: MenuDao
) {

    fun observarMenuDelDia(
        fecha: String
    ): Flow<List<MenuItem>> {
        return dao.observarPorFecha(fecha)
    }

    suspend fun guardar(item: MenuItem) {
        if (item.id == 0) {
            dao.insertar(item)
        } else {
            dao.actualizar(item)
        }
    }

    suspend fun eliminar(item: MenuItem) {
        dao.eliminar(item)
    }

    suspend fun cambiarDisponibilidad(
        id: Int,
        disponible: Boolean
    ) {
        dao.actualizarDisponibilidad(id, disponible)
    }
}

import androidx.lifecycle.ViewModel

import com.example.casajojo.data.model.MenuItem
import com.example.casajojo.data.repository.MenuRepository
import java.time.LocalDate
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharingStarted
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.stateIn
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

data class MenuFormState(
    val id: Int? = null,
    val nombre: String = "",
    val descripcion: String = "",
    val precio: String = "",
    val disponible: Boolean = true,
    val imagenUri: String? = null,
    val fechaMenu: String = LocalDate.now().toString(),
    val errores: MenuFormErrors = MenuFormErrors()
)

class AdminViewModel(
    private val repository: MenuRepository
) : ViewModel() {

    private val fechaHoy = LocalDate.now().toString()

    val menuItems: StateFlow<List<MenuItem>> =
        repository.observarMenuDelDia(fechaHoy).stateIn(
            scope = viewModelScope,
            started = SharingStarted.WhileSubscribed(5000),
            initialValue = emptyList()
        )

    private val _form = MutableStateFlow(
        MenuFormState(fechaMenu = fechaHoy)
    )
    val form: StateFlow<MenuFormState> = _form.asStateFlow()

    private val _mensaje = MutableStateFlow<String?>(null)
    val mensaje: StateFlow<String?> = _mensaje.asStateFlow()

    fun nuevoPlato() {
        _form.value = MenuFormState(fechaMenu = fechaHoy)
    }

    fun cargarPlato(item: MenuItem) {
        _form.value = MenuFormState(
            id = item.id,
            nombre = item.nombre,
            descripcion = item.descripcion,
            precio = item.precio.toString(),
            disponible = item.disponible,
            imagenUri = item.imagenUri,
            fechaMenu = item.fechaMenu
        )
    }

    fun onNombreChange(valor: String) {
        _form.update {
            it.copy(
                nombre = valor,
                errores = it.errores.copy(nombre = null)
            )
        }
    }

            nombre = actual.nombre,
            descripcion = actual.descripcion,
            precio = actual.precio
        )

        if (errores.tieneErrores) {
            _form.update { it.copy(errores = errores) }
            return false
        }

        val item = MenuItem(
            id = actual.id ?: 0,
            nombre = actual.nombre.trim(),
            descripcion = actual.descripcion.trim(),
            precio = actual.precio.toInt(),
            disponible = actual.disponible,
            imagenUri = actual.imagenUri,
            fechaMenu = actual.fechaMenu
        )

        viewModelScope.launch {
            repository.guardar(item)

            _mensaje.value = if (actual.id == null) {
                "Plato agregado al menÃº."
            } else {
                "Cambios guardados correctamente."
            }

            _form.value = MenuFormState(fechaMenu = fechaHoy)
        }

        return true
    }

        }
    }

    fun eliminarPlato(item: MenuItem) {
        viewModelScope.launch {
            repository.eliminar(item)

        }
    }

        _mensaje.value = null
    }

        _checkout.update {
            it.copy(
                restricciones = valor,
                errores = it.errores.copy(restricciones = null)
            )
        }
    }

    fun confirmarPedido() {
        val estado = _checkout.value
        if (estado.guardando) return

        val carritoActual = _carrito.value
        val idsDisponibles = menuDisponible.value
            .orEmpty()
            .map { it.id }
            .toSet()

        val errores = PedidoValidator.validarCheckout(
            nombre = estado.nombre,
            retiroLoAntesPosible = estado.retiroLoAntesPosible,
            horaRetiro = estado.horaRetiro,
            restricciones = estado.restricciones,
            carrito = carritoActual,
            idsDisponibles = idsDisponibles,
            horaActual = LocalTime.now()
        )

        if (errores.tieneErrores) {
            _checkout.update { it.copy(errores = errores, errorGuardado = null) }
            return
        }

        val pedido = Pedido(
            clienteNombre = estado.nombre.trim(),
            detalleItems = carritoActual.aTextoDetalle(),
            restricciones = estado.restricciones.trim(),
            horaComprometida = if (estado.retiroLoAntesPosible) {
                PedidoRepository.RETIRO_LO_ANTES_POSIBLE
            } else {
                estado.horaRetiro.orEmpty()
            },
            total = carritoActual.calcularTotal()
        )

        _checkout.update {
            it.copy(
                guardando = true,
                errores = CheckoutErrores(),
                errorGuardado = null
            )
        }

        viewModelScope.launch {
            try {
                val idNuevo = pedidoRepository.registrar(pedido)

                _pedidoSeleccionadoId.value = idNuevo
                _carrito.value = emptyList()
                _checkout.value = CheckoutUiState(pedidoConfirmadoId = idNuevo)
            } catch (e: CancellationException) {
                throw e
            } catch (e: Exception) {
                _checkout.update {
                    it.copy(
                        guardando = false,
                        errorGuardado = "No se pudo guardar tu pedido. " +
                                "Tu carrito se mantiene; intenta nuevamente."
                    )
                }
            }
        }
    }

    fun confirmacionAtendida() {
        _checkout.update { it.copy(pedidoConfirmadoId = null) }
    }

    fun seleccionarPedido(id: Int) {
        _pedidoSeleccionadoId.value = id
    }

private fun limpiarErrorCarrito() {
        _checkout.update { it.copy(errores = it.errores.copy(carrito = null)) }
    }

    private fun mostrarMensajeTemporal(texto: String) {
        trabajoMensaje?.cancel()
        _mensaje.value = texto
        _mostrarMensaje.value = true
        trabajoMensaje = viewModelScope.launch {
            delay(2000)
            _mostrarMensaje.value = false
        }
    }
}

class ClienteViewModelFactory(

        if (modelClass.isAssignableFrom(ClienteViewModel::class.java)) {
            @Suppress("UNCHECKED_CAST")
            return ClienteViewModel(menuRepository, pedidoRepository) as T
        }
        throw IllegalArgumentException("ViewModel desconocido: ${modelClass.name}")
    }

private fun fechaLegible(fechaIso: String): String {
    return try {
        val formato = DateTimeFormatter.ofPattern("EEEE d 'de' MMMM", LOCALE_CHILE)
